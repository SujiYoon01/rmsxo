# -*- coding: utf-8 -*- """ 근무시간 분석 자동화 스크립트 ------------------------------------------------ 사용법: python 근무시간_분석.py 로우데이터.xlsx 입력: - 엑셀(xlsx) 원본 파일 1개 (A~V열 구조, 헤더는 1행) 출력: - dashboard.html (같은 폴더에 생성됨. 더블클릭해서 브라우저로 열면 됨) * 인터넷 연결이 있는 브라우저에서 열어야 그래프가 정상적으로 보입니다. (Chart.js 라이브러리를 CDN에서 불러오기 때문) """ import sys import json import pandas as pd # ------------------------------------------------------------------ # 0. 설정값 (필요하면 여기만 수정하면 됩니다) # ------------------------------------------------------------------ INPUT_PATH = sys.argv[1] if len(sys.argv) > 1 else "로우데이터.xlsx" OUTPUT_PATH = "dashboard.html" EXCLUDE_구분 = ["근태예외자"] # 분석에서 제외할 '구분' 값 DEPT_REMAP = { # 부서명 통합 규칙 "제조팀": "제주우주센터", "체계기술2팀": "제주우주센터", } # ------------------------------------------------------------------ # 1. 데이터 로드 & 전처리 # ------------------------------------------------------------------ def load_and_clean(path: str) -> pd.DataFrame: df = pd.read_excel(path) # 1) 근태예외자 제외 df = df[~df["구분"].isin(EXCLUDE_구분)].copy() # 2) 부서명 통합 (제조팀, 체계기술2팀 -> 제주우주센터) df["부서명"] = df["부서명"].replace(DEPT_REMAP) # 3) 주확정근무시간: 분 -> 시간, 소수점 첫째자리 df["주확정근무시간_시간"] = (df["주확정근무시간"] / 60).round(1) return df # ------------------------------------------------------------------ # 2. 개인별 월평균 근무시간 (사번 기준으로 1~4주차 평균) # ------------------------------------------------------------------ def build_personal_avg(df: pd.DataFrame) -> pd.DataFrame: personal = ( df.groupby(["사번", "이름", "부서명"], as_index=False)["주확정근무시간_시간"] .mean() .rename(columns={"주확정근무시간_시간": "월평균근무시간"}) ) personal["월평균근무시간"] = personal["월평균근무시간"].round(1) return personal # ------------------------------------------------------------------ # 3. 부서별 통계 (팀별 비교용 막대그래프 + 4사분면용) # ------------------------------------------------------------------ def build_dept_stats(personal: pd.DataFrame) -> pd.DataFrame: dept = ( personal.groupby("부서명")["월평균근무시간"] .agg(부서평균="mean", 부서표준편차="std", 인원수="count") .reset_index() ) # 인원이 1명뿐인 부서는 표준편차가 NaN으로 나옴 -> 0으로 처리 dept["부서표준편차"] = dept["부서표준편차"].fillna(0) dept["부서평균"] = dept["부서평균"].round(1) dept["부서표준편차"] = dept["부서표준편차"].round(2) return dept.sort_values("부서평균", ascending=False) # ------------------------------------------------------------------ # 4. HTML 대시보드 생성 # ------------------------------------------------------------------ HTML_TEMPLATE = """ 근무시간 분석 대시보드

근무시간 분석 대시보드

1. 부서별 평균 근무시간 비교

2. 근무시간 4사분면 (X: 부서 평균 근무시간, Y: 부서 내 편차)

점선은 전체 부서 평균값 기준입니다. 우측 상단(1사분면)일수록 해당 부서의 평균 근무시간이 길고, 동시에 부서 내 인원들 간 근무시간 편차도 크다는 뜻입니다(= 부서 내 업무량이 특정 인원에게 쏠렸을 가능성).

3. 개인별 근무시간

사번 이름 부서명 월평균근무시간(시간)
""" def build_html(dept_stats: pd.DataFrame, personal: pd.DataFrame) -> str: html = HTML_TEMPLATE html = html.replace("__DEPT_STATS__", json.dumps(dept_stats.to_dict("records"), ensure_ascii=False)) html = html.replace("__PERSONAL_DATA__", json.dumps(personal.to_dict("records"), ensure_ascii=False)) return html # ------------------------------------------------------------------ # main # ------------------------------------------------------------------ def main(): df = load_and_clean(INPUT_PATH) personal = build_personal_avg(df) dept_stats = build_dept_stats(personal) html = build_html(dept_stats, personal) with open(OUTPUT_PATH, "w", encoding="utf-8") as f: f.write(html) print(f"완료: {OUTPUT_PATH} 생성됨 (부서 {len(dept_stats)}개, 인원 {len(personal)}명)") if __name__ == "__main__": main()
